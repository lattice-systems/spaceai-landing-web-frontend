# Home Hero Implementation Plan

> **For agentic workers:** Use superpowers:executing-plans to implement task-by-task.

**Goal:** Construir la sección hero del home con layout split (texto izquierda, campus-graph SVG derecha), Angular fade-in animation, colores del tema, sin gradientes.

**Architecture:** Hero standalone component que compone `CampusGraph` (SVG puro con CSS keyframe animations). `Home` page importa `Hero`. Sin estado externo ni servicios.

**Tech Stack:** Angular 22, Tailwind CSS v4, `@angular/animations`, spartan `HlmButtonImports`, `@ng-icons/lucide`

## Global Constraints

- Standalone components, sin `standalone: true` en decorator (default Angular v20+)
- `ChangeDetectionStrategy.OnPush` en todos los componentes
- Sin gradientes — solo colores del theme (`text-primary`, `bg-background`, `text-foreground`, etc.)
- Colores SVG via `currentColor` + clases Tailwind
- Sin assets externos — todo SVG inline
- WCAG AA: contraste mínimo 4.5:1 en texto, `aria-hidden="true"` en SVG decorativo

```typescript
import { ChangeDetectionStrategy, Component } from '@angular/core';

const NODES = [
  { id: 'mobile',   label: 'App Móvil',        x: 80,  y: 60  },
  { id: 'access',   label: 'Control Acceso',   x: 320, y: 60  },
  { id: 'kiosk',    label: 'Kiosco SIDE',      x: 80,  y: 240 },
  { id: 'robot',    label: 'Robot Autónomo',   x: 320, y: 240 },
] as const;

const CENTER = { x: 200, y: 150 };

@Component({
  selector: 'app-campus-graph',
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    <svg
      viewBox="0 0 400 300"
      aria-hidden="true"
      class="w-full max-w-lg"
      xmlns="http://www.w3.org/2000/svg"
    >
      <style>
        .graph-line {
          stroke-dasharray: 6 4;
          animation: dash 2s linear infinite;
        }
        .graph-line:nth-child(2) { animation-delay: -0.5s; }
        .graph-line:nth-child(3) { animation-delay: -1s; }
        .graph-line:nth-child(4) { animation-delay: -1.5s; }
        .pulse-dot {
          animation: pulse-dot 2s ease-in-out infinite;
        }
        .pulse-dot:nth-child(2) { animation-delay: -0.5s; }
        .pulse-dot:nth-child(3) { animation-delay: -1s; }
        .pulse-dot:nth-child(4) { animation-delay: -1.5s; }
        @keyframes dash {
          to { stroke-dashoffset: -20; }
        }
        @keyframes pulse-dot {
          0%, 100% { opacity: 0.3; r: 3; }
          50%       { opacity: 1;   r: 5; }
        }
      </style>

      <!-- Lines center → node -->
      @for (node of nodes; track node.id) {
        <line
          class="graph-line"
          [attr.x1]="center.x" [attr.y1]="center.y"
          [attr.x2]="node.x"   [attr.y2]="node.y"
          stroke="currentColor"
          stroke-width="1.5"
          class="text-primary opacity-40 graph-line"
        />
      }

      <!-- Pulse dots mid-line -->
      @for (node of nodes; track node.id) {
        <circle
          class="pulse-dot"
          [attr.cx]="(center.x + node.x) / 2"
          [attr.cy]="(center.y + node.y) / 2"
          r="3"
          fill="currentColor"
          class="text-primary"
        />
      }

      <!-- Satellite nodes -->
      @for (node of nodes; track node.id) {
        <g>
          <circle
            [attr.cx]="node.x" [attr.cy]="node.y" r="28"
            fill="currentColor" class="text-primary opacity-10"
            stroke="currentColor" stroke-width="1"
            class="text-primary opacity-40"
          />
          <text
            [attr.x]="node.x" [attr.y]="node.y - 6"
            text-anchor="middle"
            font-size="9"
            fill="currentColor"
            class="text-foreground"
            font-weight="500"
          >{{ node.label.split(' ')[0] }}</text>
          <text
            [attr.x]="node.x" [attr.y]="node.y + 7"
            text-anchor="middle"
            font-size="9"
            fill="currentColor"
            class="text-foreground"
          >{{ node.label.split(' ').slice(1).join(' ') }}</text>
        </g>
      }

      <!-- Center node -->
      <circle
        [attr.cx]="center.x" [attr.cy]="center.y" r="38"
        fill="currentColor" class="text-primary opacity-15"
        stroke="currentColor" stroke-width="1.5"
        class="text-primary"
      />
      <text
        [attr.x]="center.x" [attr.y]="center.y - 5"
        text-anchor="middle"
        font-size="13"
        font-weight="700"
        fill="currentColor"
        class="text-foreground"
      >Space</text>
      <text
        [attr.x]="center.x" [attr.y]="center.y + 10"
        text-anchor="middle"
        font-size="13"
        font-weight="700"
        fill="currentColor"
        class="text-primary"
      >IA</text>
    </svg>
  `,
})
export class CampusGraph {
  protected readonly nodes = NODES;
  protected readonly center = CENTER;
}
```

```typescript
import { animate, style, transition, trigger } from '@angular/animations';
import { ChangeDetectionStrategy, Component } from '@angular/core';
import { RouterLink } from '@angular/router';
import { HlmButtonImports } from '@spartan-ng/helm/button';
import { CampusGraph } from './campus-graph/campus-graph';

@Component({
  selector: 'app-hero',
  imports: [RouterLink, HlmButtonImports, CampusGraph],
  changeDetection: ChangeDetectionStrategy.OnPush,
  animations: [
    trigger('fadeSlideIn', [
      transition(':enter', [
        style({ opacity: 0, transform: 'translateY(20px)' }),
        animate('500ms ease-out', style({ opacity: 1, transform: 'translateY(0)' })),
      ]),
    ]),
  ],
  template: `
    <section class="mx-auto flex min-h-[calc(100vh-4rem)] max-w-7xl items-center px-4 py-20 sm:px-6">
      <div class="grid w-full items-center gap-12 md:grid-cols-2">

        <!-- Left: copy -->
        <div @fadeSlideIn class="flex flex-col gap-6">
          <span class="w-fit rounded-full border border-border px-3 py-1 text-xs font-medium text-muted-foreground">
            Ecosistema Universitario Inteligente
          </span>

          <h1 class="text-4xl font-bold leading-tight tracking-tight text-foreground sm:text-5xl lg:text-6xl">
            Transforma tu institución con el
            <span class="text-primary"> campus inteligente</span>
          </h1>

          <p class="max-w-md text-base leading-relaxed text-muted-foreground">
            SpaceIA integra IA, IoT y robótica para automatizar procesos,
            modernizar espacios y elevar la experiencia de cada estudiante.
          </p>

          <div class="flex flex-wrap gap-3">
            <a hlmBtn size="lg" routerLink="/cotizador">Solicitar cotización</a>
            <a hlmBtn size="lg" variant="ghost" routerLink="/spaceai">Conocer productos</a>
          </div>
        </div>

        <!-- Right: campus graph -->
        <div @fadeSlideIn class="flex items-center justify-center">
          <app-campus-graph />
        </div>

      </div>
    </section>
  `,
})
export class Hero {}
```
